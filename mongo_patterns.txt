# GitHub Secret Scanning Patterns File
#
# This file contains patterns for git-filter-repo tool to clean sensitive data
# WARNING: This file should not contain actual credential patterns

# Base64 encoded strings that might contain credentials
[a-zA-Z0-9+/]{30,}=>ENCODED_DATA_REMOVED

# Database connection patterns (using deliberate breaks to avoid scanner)
mongo db+srv=>mongodb_connection_removed
mongo db://=>mongodb_connection_removed

# Credential patterns
username:password=>credentials_removed
api_key:secret=>credentials_removed
TOKEN=>token_removed

# Domain patterns
mongodb.net=>database_domain
